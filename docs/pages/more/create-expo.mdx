import React, { useState, useEffect } from "react";
import { View, Text, TextInput, ScrollView, Button, StyleSheet } from "react-native";
import AsyncStorage from "@react-native-async-storage/async-storage";

export default function App() {
  const [data, setData] = useState({
    walking: "",
    prostrations: "",
    writing: "",
    recording: "",
    tebel: "",
    keto: "",
    eatingWindow: "",
    manaContent: "",
    statusContent: "",
    manaService: "",
    statusService: ""
  });

  const todayKey = new Date().toISOString().split("T")[0];

  useEffect(() => {
    loadData();
  }, []);

  const loadData = async () => {
    const saved = await AsyncStorage.getItem(todayKey);
    if (saved) setData(JSON.parse(saved));
  };

  const saveData = async () => {
    await AsyncStorage.setItem(todayKey, JSON.stringify(data));
    alert("Saved");
  };

  const update = (key, value) => {
    setData({ ...data, [key]: value });
  };

  return (
    <ScrollView style={styles.container}>
      <Text style={styles.header}>Daily Discipline Tracker</Text>

      <Input label="Walking (km)" value={data.walking} onChange={(v)=>update("walking", v)} />
      <Input label="Prostrations" value={data.prostrations} onChange={(v)=>update("prostrations", v)} />
      <Input label="Writing (words)" value={data.writing} onChange={(v)=>update("writing", v)} />
      <Input label="Recording (minutes)" value={data.recording} onChange={(v)=>update("recording", v)} />
      <Input label="Tebel (litres)" value={data.tebel} onChange={(v)=>update("tebel", v)} />
      <Input label="Eating Window (hrs)" value={data.eatingWindow} onChange={(v)=>update("eatingWindow", v)} />
      <Input label="Keto (Yes/No)" value={data.keto} onChange={(v)=>update("keto", v)} />

      <Text style={styles.sub}>Created Content</Text>
      <Input label="Daily Mana" value={data.manaContent} onChange={(v)=>update("manaContent", v)} />
      <Input label="Performance Status" value={data.statusContent} onChange={(v)=>update("statusContent", v)} />

      <Text style={styles.sub}>Other Services</Text>
      <Input label="Daily Mana" value={data.manaService} onChange={(v)=>update("manaService", v)} />
      <Input label="Performance Status" value={data.statusService} onChange={(v)=>update("statusService", v)} />

      <Button title="Save Day" onPress={saveData} />
    </ScrollView>
  );
}

const Input = ({ label, value, onChange }) => (
  <View style={{ marginBottom: 10 }}>
    <Text>{label}</Text>
    <TextInput
      style={styles.input}
      value={value}
      onChangeText={onChange}
    />
  </View>
);

const styles = StyleSheet.create({
  container: { padding: 20 },
  header: { fontSize: 22, fontWeight: "bold", marginBottom: 20 },
  sub: { fontSize: 18, marginTop: 20 },
  input: {
    borderWidth: 1,
    padding: 8,
    borderRadius: 5
  }
});
